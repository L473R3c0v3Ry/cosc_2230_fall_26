Make =ThreeSum= quadratic by trading memory for time. About 90 to 120
minutes.
The idea: if all values lie in [-M, M), keep a table with one slot per
possible value; the value itself is the index, so a lookup costs one
array access.

Work in a folder with your =ThreeSum.java=, =Stopwatch.java=,
=stdlib.jar= and these classes:

#+begin_src java :eval never :tangle ../src/Distinct.java
  // Distinct.java: n distinct random ints in [-M, M)
  public class Distinct
  {
      public static final int M = 1000000;
      public static int[] randomArray(int n)
      {
          boolean[] used = new boolean[2 * M];   // used[x + M]: x is taken
          int[] a = new int[n];
          for (int i = 0; i < n; i++)
          {
              int x;
              do x = StdRandom.uniformInt(-M, M); while (used[x + M]);
              used[x + M] = true;
              a[i] = x;
          }
          return a;
      }
  }
#+end_src

#+begin_src java :eval never :tangle ../src/TwoSum.java
  // TwoSum.java: brute force, counts pairs that sum to zero
  public class TwoSum
  {
      public static int countPairs(int[] a)
      {
          int n = a.length, count = 0;
          for (int i = 0; i < n; i++)
              for (int j = i + 1; j < n; j++)
                  if (a[i] + a[j] == 0) count++;
          return count;
      }
  }
#+end_src

  {

      {

      public static void main(String[] args)
      {
          int max = Integer.parseInt(args[0]);
          double previous = timeTrial(1000);
